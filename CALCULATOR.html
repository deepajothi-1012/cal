// server.js const express = require('express'); const bodyParser = require('body-parser'); const app = express(); const PORT = 3000; // Middleware app.use(bodyParser.json()); // Serve the calculator HTML directly app.get('/', (req, res) => { res.send(` Calculator

Simple Calculator

`); }); // Calculate result app.post('/calculate', (req, res) => { try { const { expression } = req.body; const result = eval(expression); // simple evaluation res.json({ result }); } catch (err) { res.json({ result: 'Error' }); } }); // Start server app.listen(PORT, () => { console.log(`Calculator running at http://localhost:${PORT}`); });
